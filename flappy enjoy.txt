<script>
(() => {
  'use strict';

  const $ = id => document.getElementById(id);
  const canvas = $('canvas');
  const ctx = canvas.getContext('2d');
  const W = 420, H = 680, FLOOR = 608;
  const BIRD_X = 106, RADIUS = 13;
  const STEP = 1 / 120;
  const STORAGE_KEY = 'little-flight-best-v1';
  const SOUND_KEY = 'little-flight-sound-v1';

  function readStorage(key, fallback) {
    try { return localStorage.getItem(key) ?? fallback; }
    catch { return fallback; }
  }
  function writeStorage(key, value) {
    try { localStorage.setItem(key, String(value)); }
    catch { /* Gameplay remains available when storage is blocked. */ }
  }

  let best = Number(readStorage(STORAGE_KEY, '0'));
  if (!Number.isSafeInteger(best) || best < 0) best = 0;
  let soundEnabled = readStorage(SOUND_KEY, 'on') !== 'off';
  let audio;
  let state = 'ready';
  let score = 0, pipes = [], birdY = 310, velocity = 0;
  let distance = 0, spawnDistance = 0, elapsed = 0;
  let lastTime = 0, accumulator = 0, flash = 0, deathTime = 0;
  let wingKick = 0;

  const reducedMotion = matchMedia('(prefers-reduced-motion: reduce)').matches;
  $('best').textContent = best;

  function unlockAudio() {
    if (!soundEnabled) return;
    try {
      const AudioContext = window.AudioContext || window.webkitAudioContext;
      if (!AudioContext) return;
      if (!audio) audio = new AudioContext();
      if (audio.state === 'suspended') audio.resume().catch(() => {});
    } catch { /* Audio is optional. */ }
  }

  function tone(from, to, duration, volume = .055, type = 'sine', delay = 0) {
    if (!soundEnabled || !audio || audio.state !== 'running') return;
    const now = audio.currentTime + delay;
    const oscillator = audio.createOscillator();
    const gain = audio.createGain();
    oscillator.type = type;
    oscillator.frequency.setValueAtTime(from, now);
    oscillator.frequency.exponentialRampToValueAtTime(to, now + duration);
    gain.gain.setValueAtTime(.0001, now);
    gain.gain.exponentialRampToValueAtTime(volume, now + .012);
    gain.gain.exponentialRampToValueAtTime(.0001, now + duration);
    oscillator.connect(gain);
    gain.connect(audio.destination);
    oscillator.start(now);
    oscillator.stop(now + duration + .02);
    oscillator.onended = () => { oscillator.disconnect(); gain.disconnect(); };
  }

  function updateSoundButton() {
    $('sound').setAttribute('aria-pressed', String(soundEnabled));
    $('sound').setAttribute('aria-label', soundEnabled ? 'Mute sound' : 'Enable sound');
    $('soundWaves').style.display = soundEnabled ? '' : 'none';
    $('soundSlash').style.display = soundEnabled ? 'none' : '';
  }

  $('sound').addEventListener('click', () => {
    soundEnabled = !soundEnabled;
    writeStorage(SOUND_KEY, soundEnabled ? 'on' : 'off');
    updateSoundButton();
    unlockAudio();
    if (!soundEnabled && audio) audio.suspend().catch(() => {});
  });
  updateSoundButton();

  function showScreen(screen) {
    $('startScreen').hidden = screen !== 'start';
    $('overScreen').hidden = screen !== 'over';
    $('pauseScreen').hidden = screen !== 'pause';
  }

  function flap() {
    velocity = -350;
    wingKick = 1;
    tone(430, 780, .085, .035);
  }

  function start() {
    unlockAudio();
    state = 'playing';
    score = 0;
    pipes = [];
    birdY = 285;
    velocity = 0;
    distance = 0;
    spawnDistance = 0;
    flash = 0;
    accumulator = 0;
    lastTime = 0;
    $('score').textContent = '0';
    showScreen(null);
    $('announcement').textContent = 'Flight started.';
    // The first obstacle begins offscreen, leaving time to establish a rhythm.
    addPipe(W + 110);
    flap();
  }

  function pause() {
    if (state !== 'playing') return;
    state = 'paused';
    showScreen('pause');
    $('announcement').textContent = 'Flight paused.';
  }

  function resume() {
    if (state !== 'paused') return;
    unlockAudio();
    state = 'playing';
    lastTime = 0;
    accumulator = 0;
    showScreen(null);
    flap();
  }

  function end() {
    if (state !== 'playing') return;
    state = 'over';
    deathTime = performance.now();
    flash = reducedMotion ? 0 : .22;
    const newRecord = score > best;
    if (newRecord) {
      best = score;
      writeStorage(STORAGE_KEY, best);
    }
    $('best').textContent = best;
    $('finalScore').textContent = score;
    $('finalBest').textContent = best;
    $('overEyebrow').textContent = newRecord ? 'A new personal best!' : 'Every flight is a fresh start';
    $('overEyebrow').classList.toggle('record', newRecord);
    $('overMessage').textContent = score === 0
      ? 'Short, gentle taps. Aim for the middle of each gap.'
      : newRecord ? 'A little further than ever before. Beautiful.'
      : 'Take a breath. The sky will wait.';
    showScreen('over');
    $('announcement').textContent = `Game over. Score ${score}. Personal best ${best}.`;
    tone(170, 45, .26, .08, 'triangle');
  }

  function action() {
    unlockAudio();
    if (state === 'ready') start();
    else if (state === 'playing') flap();
    else if (state === 'paused') resume();
    else if (performance.now() - deathTime > 350) start();
  }

  $('startButton').addEventListener('click', start);
  $('restartButton').addEventListener('click', start);
  $('resumeButton').addEventListener('click', resume);

  $('game').addEventListener('pointerdown', event => {
    if (event.target.closest('button')) return;
    if (event.pointerType === 'mouse' && event.button !== 0) return;
    if (!event.isPrimary) return;
    event.preventDefault();
    // Game-over restarts use the explicit button to avoid accidental retries.
    if (state !== 'over') action();
  });

  document.addEventListener('keydown', event => {
    if (event.code === 'Escape' || event.code === 'KeyP') {
      event.preventDefault();
      if (event.repeat) return;
      if (state === 'playing') pause();
      else if (state === 'paused') resume();
      return;
    }
    if (!['Space', 'ArrowUp', 'KeyW', 'Enter'].includes(event.code)) return;
    if (event.target.closest('button') && ['Space', 'Enter'].includes(event.code)) return;
    event.preventDefault();
    if (!event.repeat) action();
  });

  window.addEventListener('blur', pause);
  document.addEventListener('visibilitychange', () => {
    if (document.hidden) pause();
  });

  function difficulty() { return Math.min(score / 35, 1); }

  function addPipe(x) {
    const level = difficulty();
    const gap = 177 - level * 39;
    const previous = pipes[pipes.length - 1];
    let center = 190 + Math.random() * 235;
    if (previous) {
      center = Math.max(previous.center - 115, Math.min(previous.center + 115, center));
    }
    pipes.push({ x, center, gap, width: 66, scored: false });
  }

  function circleHitsRect(cx, cy, r, x, y, width, height) {
    const dx = cx - Math.max(x, Math.min(cx, x + width));
    const dy = cy - Math.max(y, Math.min(cy, y + height));
    return dx * dx + dy * dy < r * r;
  }

  function update(dt) {
    elapsed += dt;
    wingKick = Math.max(0, wingKick - dt * 5);
    flash = Math.max(0, flash - dt);

    if (state === 'ready') {
      if (!reducedMotion) distance += 25 * dt;
      return;
    }

    if (state === 'over') {
      velocity = Math.min(velocity + 1150 * dt, 700);
      birdY = Math.min(FLOOR - RADIUS, birdY + velocity * dt);
      return;
    }

    if (state !== 'playing') return;

    const speed = 155 + 65 * difficulty();
    const travel = speed * dt;
    distance += travel;
    spawnDistance += travel;
    velocity = Math.min(velocity + 1120 * dt, 650);
    birdY += velocity * dt;

    if (spawnDistance >= 235) {
      spawnDistance -= 235;
      addPipe(W + 110 - spawnDistance);
    }

    for (const pipe of pipes) {
      pipe.x -= travel;
      const top = pipe.center - pipe.gap / 2;
      const bottom = pipe.center + pipe.gap / 2;

      // Collision bounds include the slightly wider pipe lips.
      if (
        circleHitsRect(BIRD_X, birdY, RADIUS, pipe.x, 0, pipe.width, top) ||
        circleHitsRect(BIRD_X, birdY, RADIUS, pipe.x - 4, top - 24, pipe.width + 8, 24) ||
        circleHitsRect(BIRD_X, birdY, RADIUS, pipe.x, bottom, pipe.width, FLOOR - bottom) ||
        circleHitsRect(BIRD_X, birdY, RADIUS, pipe.x - 4, bottom, pipe.width + 8, 24)
      ) {
        end();
        return;
      }

      if (!pipe.scored && pipe.x + pipe.width + 4 < BIRD_X - RADIUS) {
        pipe.scored = true;
        score++;
        $('score').textContent = score;
        tone(660, 880, .12, .05);
        tone(990, 1320, .16, .04, 'sine', .09);
      }
    }

    pipes = pipes.filter(pipe => pipe.x + pipe.width > -10);

    if (birdY - RADIUS <= 0 || birdY + RADIUS >= FLOOR) {
      birdY = Math.max(RADIUS, Math.min(FLOOR - RADIUS, birdY));
      end();
    }
  }

  function roundRect(x, y, w, h, r, fill) {
    ctx.fillStyle = fill;
    ctx.beginPath();
    ctx.roundRect(x, y, w, h, r);
    ctx.fill();
  }

  function cloud(x, y, scale, alpha) {
    ctx.save();
    ctx.translate(x, y);
    ctx.scale(scale, scale);
    ctx.fillStyle = `rgba(255,250,230,${alpha})`;
    ctx.beginPath();
    ctx.ellipse(0, 0, 36, 13, 0, 0, Math.PI * 2);
    ctx.ellipse(-15, -10, 17, 17, 0, 0, Math.PI * 2);
    ctx.ellipse(8, -15, 22, 21, 0, 0, Math.PI * 2);
    ctx.ellipse(30, -3, 20, 13, 0, 0, Math.PI * 2);
    ctx.fill();
    ctx.restore();
  }

  function hills(base, color, amplitude, period, offset) {
    ctx.fillStyle = color;
    ctx.beginPath();
    ctx.moveTo(0, FLOOR);
    for (let x = 0; x <= W + 8; x += 8) {
      const y = base + Math.sin((x + offset) / period) * amplitude
        + Math.sin((x + offset) / (period * .53)) * amplitude * .25;
      ctx.lineTo(x, y);
    }
    ctx.lineTo(W, FLOOR);
    ctx.closePath();
    ctx.fill();
  }

  function drawPipe(pipe) {
    const top = pipe.center - pipe.gap / 2;
    const bottom = pipe.center + pipe.gap / 2;
    const gradient = ctx.createLinearGradient(pipe.x, 0, pipe.x + pipe.width, 0);
    gradient.addColorStop(0, '#397e76');
    gradient.addColorStop(.3, '#60a08b');
    gradient.addColorStop(1, '#34736e');

    for (const [y, height, lipY] of [
      [-10, top + 10, top - 24],
      [bottom, FLOOR - bottom, bottom]
    ]) {
      roundRect(pipe.x, y, pipe.width, height, 5, gradient);
      ctx.fillStyle = '#c5dfb92b';
      ctx.fillRect(pipe.x + 10, y, 5, height);
      ctx.fillStyle = '#123e4820';
      ctx.fillRect(pipe.x + pipe.width - 9, y, 5, height);
      roundRect(pipe.x - 4, lipY + 3, pipe.width + 8, 24, 5, '#245e5a');
      roundRect(pipe.x - 4, lipY, pipe.width + 8, 22, 5, '#71a68c');
      ctx.fillStyle = '#b9d7a780';
      ctx.fillRect(pipe.x + 1, lipY + 3, pipe.width - 2, 3);
    }

    ctx.fillStyle = '#e6e9bb75';
    const markY = bottom + 41;
    for (let i = 0; i < 3; i++) {
      ctx.beginPath();
      ctx.arc(pipe.x + pipe.width / 2, markY + i * 9, 1.6, 0, Math.PI * 2);
      ctx.fill();
    }
  }

  function drawBird(x, y, rotation, time) {
    ctx.save();
    ctx.translate(x, y);
    ctx.rotate(rotation);

    // Tail feathers.
    ctx.fillStyle = '#b95837';
    ctx.beginPath();
    ctx.moveTo(-15, 3);
    ctx.lineTo(-29, -8);
    ctx.lineTo(-26, 8);
    ctx.lineTo(-13, 11);
    ctx.fill();

    ctx.fillStyle = '#ef794d';
    ctx.beginPath();
    ctx.ellipse(0, 0, 21, 18, -.1, 0, Math.PI * 2);
    ctx.fill();

    ctx.fillStyle = '#ffbe76';
    ctx.beginPath();
    ctx.ellipse(4, 7, 15, 10, -.15, 0, Math.PI * 2);
    ctx.fill();

    const wingAngle = state === 'over' ? .5
      : Math.sin(time * 22) * .35 - wingKick * .8;
    ctx.save();
    ctx.translate(-7, 3);
    ctx.rotate(wingAngle);
    ctx.fillStyle = '#c95b3d';
    ctx.beginPath();
    ctx.ellipse(-3, 2, 12, 7, -.4, 0, Math.PI * 2);
    ctx.fill();
    ctx.restore();

    ctx.fillStyle = '#fff8e9';
    ctx.beginPath();

  function draw() {
    ctx.clearRect(0, 0, W, H);
    const sky = ctx.createLinearGradient(0, 0, 0, FLOOR);
    sky.addColorStop(0, '#acd8d8');
    sky.addColorStop(.65, '#deead1');
    sky.addColorStop(1, '#f6e9ba');
    ctx.fillStyle = sky;
    ctx.fillRect(0, 0, W, H);

    ctx.fillStyle = '#fff8df65';
    ctx.beginPath();
    ctx.arc(325, 138, 64, 0, Math.PI * 2);
    ctx.fill();
    ctx.fillStyle = '#fff5d7';
    ctx.beginPath();
    ctx.arc(325, 138, 43, 0, Math.PI * 2);
    ctx.fill();

    cloud(((90 - distance * .12) % 550 + 550) % 550 - 65, 152, 1, .56);
    cloud(((365 - distance * .08) % 570 + 570) % 570 - 65, 270, .7, .43);
    cloud(((240 - distance * .16) % 570 + 570) % 570 - 65, 355, .9, .3);

    hills(485, '#b2cdb1', 31, 90, distance * .12);
    hills(534, '#8eb8a0', 27, 69, distance * .22);
    hills(575, '#689f8c', 19, 52, distance * .36);

    for (const pipe of pipes) drawPipe(pipe);

    ctx.fillStyle = '#e6d7ab';
    ctx.fillRect(0, FLOOR, W, H - FLOOR);
    ctx.fillStyle = '#3c7e6d';
    ctx.fillRect(0, FLOOR, W, 5);
    ctx.fillStyle = '#a7bd7b';
    ctx.fillRect(0, FLOOR + 5, W, 8);
    ctx.fillStyle = '#fff3cb';
    ctx.fillRect(0, FLOOR + 13, W, 3);

    const groundOffset = distance % 28;
    ctx.strokeStyle = '#c9b98d';
    ctx.lineWidth = 2;
    for (let x = -28; x < W + 28; x += 28) {
      ctx.beginPath();
      ctx.moveTo(x - groundOffset, FLOOR + 27);
      ctx.lineTo(x - groundOffset + 10, FLOOR + 27);
      ctx.stroke();
      ctx.beginPath();
      ctx.moveTo(x - groundOffset + 14, FLOOR + 46);
      ctx.lineTo(x - groundOffset + 18, FLOOR + 46);
      ctx.stroke();
    }

    if (state === 'ready') {
      const bob = reducedMotion ? 0 : Math.sin(elapsed * 3) * 7;
      drawBird(W / 2, 388 + bob, -.08, reducedMotion ? 0 : elapsed);
    } else {
      const rotation = Math.max(-.45, Math.min(1.25, velocity / 650));
      drawBird(BIRD_X, birdY, rotation, elapsed);
    }

    if (flash > 0) {
      ctx.fillStyle = `rgba(255,248,233,${flash * 2})`;
      ctx.fillRect(0, 0, W, H);
    }
  }

  function resize() {
    const rect = canvas.getBoundingClientRect();
    const dpr = Math.min(window.devicePixelRatio || 1, 3);
    canvas.width = Math.round(rect.width * dpr);
    canvas.height = Math.round(rect.height * dpr);
    ctx.setTransform(canvas.width / W, 0, 0, canvas.height / H, 0, 0);
    draw();
  }

  function frame(now) {
    if (!lastTime) lastTime = now;
    const dt = Math.min((now - lastTime) / 1000, .05);
    lastTime = now;

    // Fixed-step simulation makes physics independent of display refresh rate.
    if (state !== 'paused' && !document.hidden) {
      accumulator += dt;
      while (accumulator >= STEP) {
        update(STEP);
        accumulator -= STEP;
      }
    }
    draw();
    requestAnimationFrame(frame);
  }

  window.addEventListener('resize', resize);
  resize();
  requestAnimationFrame(frame);
})();
